    this.socket = null;

    this.gameMode = null;

    this.attachFileHandler();

  /**
   * Initialise la connexion Socket.io au serveur
   * 
   * Établit la connexion WebSocket au serveur Planning Poker
   * et configure tous les handlers d'événements Socket
   * 
   * @function initializeSocket
   * @returns {void}
   * 
   * @description
   * - Détermine l'URL du serveur (protocol, hostname, port 3000)
   * - Crée une instance Socket.io
   * - Configure les handlers via setupSocketHandlers
   * 
   * @example
   * this.initializeSocket();
   */
  initializeSocket() {
    const protocol = window.location.protocol;
    const hostname = window.location.hostname;
    const port = 3000;
    const serverURL = `${protocol}//${hostname}:${port}`;
    this.socket = io(serverURL);
    setupSocketHandlers(this.socket, this.gameState, this);
  }

  /**
   * Crée une nouvelle session Planning Poker
   * 
   * @function createNewGame
   * @returns {void}
   * 
   * @description
   * - Récupère playerName, gameMode, participantCount du formulaire
   * - Valide que tous les champs sont remplis
   * - Envoie requête POST /api/create-session au serveur
   * - Émet 'join-session' via Socket.io
   * - Affiche l'écran de configuration
   */
  createNewGame() {
    const playerName = document.getElementById('playerName').value.trim();
    const gameMode = document.getElementById('gameMode').value;
    const participantCount = document.getElementById('participantCount').value;
    if (!playerName || !gameMode || !participantCount) {
      alert('Remplissez tous les champs');
      return;
    }
    this.playerName = playerName;
    this.gameMode = gameMode;
    this.creatorName = playerName;
    this.isRestoredSession = false;
    this.expectedParticipantCount = parseInt(participantCount);
    fetch('http://localhost:3000/api/create-session', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        creatorName: playerName,
        gameMode,
        isRestored: false,
        expectedParticipantCount: this.expectedParticipantCount
      })
    })
    .then(r => r.json())
    .then(data => {
      this.sessionCode = data.sessionCode;
      this.socket.emit('join-session', {
        sessionCode: this.sessionCode,
        playerName: this.playerName
      });
      this.showSetupScreen();
    })
    .catch(err => {
      console.error('Erreur:', err);
      alert('Impossible de créer la session');
    });
  }


  /**
   * Rejoint une session Planning Poker existante
   * 
   * @function joinGame
   * @returns {void}
   * 
   * @description
   * - Récupère playerName et sessionCode du formulaire
   * - Valide que les champs sont remplis
   * - Vérifie que la session existe
   * - Valide le nom du joueur contre la liste autorisée
   * - Émet 'join-session' via Socket.io
   * - Affiche l'écran de jeu
   */
  joinGame() {
    const playerName = document.getElementById('playerNameJoin').value.trim();
    const sessionCode = document.getElementById('sessionCode').value.trim().toUpperCase();
    if (!playerName || !sessionCode) {
      alert('Remplissez tous les champs');
      return;
    }
    this.playerName = playerName;
    this.sessionCode = sessionCode;
    fetch(`http://localhost:3000/api/session/${sessionCode}`)
      .then(r => {
        if (!r.ok) throw new Error('Session non trouvée');
        return r.json();
      })
      .then(data => {
        if (data.isRestored) {
          throw new Error('❌ Cette session est verrouillée - Partie restaurée en cours. Utilisez "Reprendre une partie" avec le fichier JSON.');
        }
        if (data.expectedParticipants && data.expectedParticipants.length > 0) {
          const exactMatch = data.expectedParticipants.some(p => p === playerName);
          if (!exactMatch) {
            throw new Error(`❌ Accès refusé: "${playerName}" ne correspond à AUCUN participant autorisé. Noms exacts attendus: ${data.expectedParticipants.join(', ')}`);
          }
        }
        this.gameMode = data.gameMode;
        this.creatorName = data.creatorName;
        this.gameState.isGameStarted = data.hasBacklog;
        this.isRestoredSession = false;
        this.expectedParticipantCount = data.expectedParticipantCount;
        this.socket.emit('join-session', {
          sessionCode: this.sessionCode,
          playerName: this.playerName,
        });
        this.showGameScreen();
      })
      .catch(err => {
        const errorEl = document.getElementById('joinError');
        errorEl.textContent = err.message || 'Session non trouvée ou terminée';
        errorEl.classList.remove('hidden');
      });
  }


  /**
   * Charge et valide un fichier JSON de session restaurée
   * 
   * @function loadResumeFile
   * @returns {void}
   * 
   * @description
   * - Récupère le fichier du formulaire
   * - Parse le JSON et valide la structure
   * - Stocke la métadonnée et les tâches
   * - Affiche un résumé du contenu chargé
   * - Affiche les participants attendus
   */
  loadResumeFile() {
    const file = document.getElementById('resumeFile').files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const data = JSON.parse(e.target.result);
        if (data.metadata && data.tasks) {
          this.restoredMetadata = data.metadata;
          this.restoredMetadata.tasks = data.tasks;
          this.restoredMetadata.completedTasks = data.completedTasks || [];
          this.restoredMetadata.taskEstimates = data.taskEstimates || {};
          this.restoredVotes = data.votes || {};
          document.getElementById('resumeFileDisplay').innerHTML = `
            <strong>✅ Fichier chargé</strong><br/>
            Participants: ${data.metadata.participantCount}<br/>
            Tâches: ${data.tasks.length}<br/>
            Dernière session: ${data.metadata.sessionCode}
          `;
          document.getElementById('resumeValidation').innerHTML = `
            <div style="background: #dbeafe; border: 1px solid #93c5fd; border-radius: 8px; padding: 12px; font-size: 13px;">
              <strong>📋 Backlog restauré</strong>
              <div style="margin-top: 8px;">
                <strong>Participants attendus (${data.metadata.participantCount}):</strong> ${data.metadata.participants.join(', ')}
              </div>
              <div style="margin-top: 8px; color: #059669;">
                ⚠️ Saisissez votre pseudo pour rejoindre la session
              </div>
            </div>
          `;
          document.getElementById('resumeStartBtn').disabled = false;
        } else {
          throw new Error('Format de fichier invalide');
        }
      } catch (err) {
        alert('Erreur: JSON invalide - ' + err.message);
      }
    };
    reader.readAsText(file);
  }


  /**
   * Démarre une session restaurée depuis un fichier
   * 
   * @function startResume
   * @returns {void}
   * 
   * @description
   * - Récupère le pseudo du joueur
   * - Valide que le pseudo est dans la liste autorisée
   * - Crée la session via API /api/create-session
   * - Émet 'join-session' via Socket.io
   * - Affiche l'écran d'attente de reconnexion
   */
  startResume() {
    const playerName = document.getElementById('playerNameResume').value.trim();
    if (!playerName) {
      alert('Saisissez votre pseudo');
      return;
    }
    if (!this.restoredMetadata || !this.restoredMetadata.participants.some(p => p === playerName)) {
      const errorEl = document.getElementById('resumeError');
      errorEl.textContent = `❌ ACCÈS REFUSÉ: "${playerName}" n'est pas dans la liste des participants. Noms exacts autorisés: ${this.restoredMetadata.participants.join(', ')}`;
      errorEl.classList.remove('hidden');
      return;
    }
    this.playerName = playerName;
    this.gameMode = this.restoredMetadata.gameMode || 'strict';
    this.creatorName = this.restoredMetadata.participants[0];
    this.isRestoredSession = true;
    this.sessionCode = this.restoredMetadata.sessionCode;
    fetch('http://localhost:3000/api/create-session', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        sessionCode: this.sessionCode,
        creatorName: this.creatorName,
        gameMode: this.gameMode,
        isRestored: true,
        expectedParticipantCount: this.restoredMetadata.participantCount,
        participants: this.restoredMetadata.participants
      })
    })
    .then(r => r.json())
    .then(data => {
      this.expectedParticipantCount = this.restoredMetadata.participantCount;
      this.socket.emit('join-session', {
        sessionCode: this.sessionCode,
        playerName: this.playerName,
      });
      this.showResumeWaitingScreen();
    })
    .catch(err => {
      console.error('Erreur:', err);
      alert('❌ Impossible de restaurer la session: ' + err.message);
    });
  }


  /**
   * Lance le jeu avec le backlog fourni
   * 
   * @function startGame
   * @returns {void}
   * 
   * @description
   * - Récupère le backlog JSON du formulaire
   * - Parse et valide le JSON
   * - Émet 'load-backlog' avec le backlog
   * - Ou affiche écran d'ajout de tâches si vide
   */
  startGame() {
    const backlogJson = document.getElementById('backlogJson').value.trim();
    if (backlogJson) {
      try {
        const backlog = JSON.parse(backlogJson);
        if (!Array.isArray(backlog)) throw new Error('Format invalide');
        this.socket.emit('load-backlog', {
          sessionCode: this.sessionCode,
          backlog: backlog,
          metadata: this.restoredMetadata,
          votes: this.restoredVotes
        });
      } catch (err) {
        alert('JSON invalide: ' + err.message);
      }
    } else {
      this.showAddTasksScreen();
    }
  }


  /**
   * Ajoute une tâche au backlog local
   * 
   * @function addTask
   * @returns {void}
   * 
   * @description
   * - Récupère taskName et taskDesc du formulaire
   * - Valide que le nom de la tâche n'est pas vide
   * - Crée un objet tâche avec id, name, description
   * - Ajoute à this.localBacklog (Array)
   * - Réinitialise le formulaire
   * - Met à jour l'affichage
   */
  addTask() {
    const name = document.getElementById('taskName').value.trim();
    const description = document.getElementById('taskDesc').value.trim();
    if (!name) {
      alert('Remplissez au moins le nom de la tâche');
      return;
    }
    const task = {
      id: this.localBacklog.length + 1,
      name: name,
      description: description
    };
    this.localBacklog.push(task);
    document.getElementById('taskName').value = '';
    document.getElementById('taskDesc').value = '';
    document.getElementById('taskName').focus();
    this.renderTasksList();
  }


  /**
   * Supprime une tâche du backlog local
   * 
   * @function removeTask
   * @param {number} index - Index de la tâche à supprimer
   * @returns {void}
   * 
   * @description
   * - Supprime l'élément à l'index donné
   * - Met à jour l'affichage
   */
  removeTask(index) {
    this.localBacklog.splice(index, 1);
    this.renderTasksList();
  }


  /**
   * Lance le vote en envoyant le backlog au serveur
   * 
   * @function startVoting
   * @returns {void}
   * 
   * @description
   * - Vérifie que le backlog n'est pas vide

      alert('Ajoutez au moins une tâche');
      return;

    this.socket.emit('load-backlog', {
      sessionCode: this.sessionCode,
      backlog: this.localBacklog
    });
  }


  /**
   * Enregistre un vote pour la tâche courante
   * 
   * @function castVote
   * @param {string|number} cardValue - Valeur du vote (ex: '8', '?', 'coffee')
   * @returns {void}
   * 
   * @description
   * - Vérifie que socket est connecté
   * - Vérifie que sessionCode existe
   * - Émet 'cast-vote' au serveur
   * - Inclut vérifications de sécurité
   */
  castVote(cardValue) {
    // ✅ VÉRIFICATION DE SÉCURITÉ
    if (!this.socket || !this.socket.connected) {
      console.error('❌ Socket non connecté');
      alert('⚠️ La connexion au serveur a été perdue. Veuillez recharger.');
      return;
    }

    if (!this.sessionCode) {
      console.error('❌ Code de session manquant');
      alert('⚠️ Code de session manquant');
      return;
    }

    console.log(`🗳️ Vote casting: ${cardValue} pour session ${this.sessionCode}`);

    // ✅ ÉMETTRE LE VOTE
    this.socket.emit('cast-vote', {
      sessionCode: this.sessionCode,
      cardValue: cardValue
    });

    console.log(`✅ Vote émis: ${cardValue}`);
  }

  }


  /**
   * Passe à la tâche suivante
   * 
   * @function continueToNext
   * @returns {void}
   * 
   * @description
   * - Marque la tâche courante comme complétée
   * - Stocke l'estimation finale (finalValue ou medianValue)
   * - Ajoute aux completedTasks
   * - Stocke l'estimation dans taskEstimates
   * - Émet 'continue-next-round' au serveur
   */
  continueToNext() {
    const currentTask = this.gameState.backlog[this.gameState.currentTaskIndex];
    if (currentTask) {
      currentTask.finalValue = this.gameState.finalValue || this.gameState.medianValue;
      currentTask.completed = true;
      currentTask.completedAt = new Date().toISOString();
      if (!this.gameState.completedTasks) {
        this.gameState.completedTasks = [];
      }
      if (!this.gameState.completedTasks.includes(currentTask.id)) {
        this.gameState.completedTasks.push(currentTask.id);
      }
      if (!this.gameState.taskEstimates) {
        this.gameState.taskEstimates = {};
      }
      this.gameState.taskEstimates[currentTask.id] = this.gameState.finalValue || this.gameState.medianValue;
    }
    this.socket.emit('continue-next-round', {
      sessionCode: this.sessionCode,
      completedTasks: this.gameState.completedTasks,
      taskEstimates: this.gameState.taskEstimates
    });
  }


  /**
   * Télécharge le backlog complet au format JSON
   * 
   * @function downloadBacklog
   * @returns {void}
   * 
   * @description
   * - Vérifie qu'il y a des tâches
   * - Enrichit les données avec métadonnées et votes
   * - Crée un blob JSON
   * - Déclenche le téléchargement du fichier
   * - Nom: backlog-[sessionCode]-[date].json
   */
  downloadBacklog() {
    if (!this.gameState.backlog || this.gameState.backlog.length === 0) {
      alert('Aucune tâche à télécharger');
      return;
    }
    const participantsList = this.gameState.players.map(p => p.name);
    if (participantsList.length !== this.expectedParticipantCount) {
      alert(`❌ Erreur: ${participantsList.length} participants connectés mais ${this.expectedParticipantCount} attendus`);
      return;
    }
    const enrichedBacklog = this.gameState.backlog.map(task => {
      if (this.gameState.completedTasks && this.gameState.completedTasks.includes(task.id)) {
        return {
          ...task,
          finalValue: this.gameState.taskEstimates[task.id] || task.finalValue,
          completed: true,
          completedAt: task.completedAt || new Date().toISOString()
        };
      }
      return task;
    });
    const backlogData = {
      metadata: {
        sessionCode: this.sessionCode,
        participantCount: participantsList.length,
        participants: participantsList,
        gameMode: this.gameMode,
        completedTasks: this.gameState.completedTasks || [],
        taskEstimates: this.gameState.taskEstimates || {},
        currentTaskIndex: this.gameState.currentTaskIndex,
        currentTaskVotes: this.gameState.currentTaskVotes || {},
        currentTaskRound: this.gameState.currentTaskRound || 1
      },
      tasks: enrichedBacklog,
      votes: this.gameState.allVotes || {}
    };
    const dataStr = JSON.stringify(backlogData, null, 2);
    const dataBlob = new Blob([dataStr], { type: 'application/json' });
    const url = URL.createObjectURL(dataBlob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `backlog-${this.sessionCode}-${new Date().toISOString().split('T')[0]}.json`;
    link.click();
    URL.revokeObjectURL(url);
  }


  /**
   * Charge et valide un fichier backlog JSON
   * 
   * @function loadBacklogFile
   * @returns {void}
   * 
   * @description
   * - Récupère le fichier du formulaire
   * - Parse le JSON
   * - Valide que metadata et tasks existent
   * - Stocke les métadonnées et votes restaurés
   * - Affiche un résumé du contenu
   */
  loadBacklogFile() {
    const file = document.getElementById('backlogFile').files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const data = JSON.parse(e.target.result);
        if (data.metadata && data.tasks) {
          if (!data.metadata.sessionCode) {
            throw new Error('Backlog invalide: sessionCode manquant (impossible de restaurer)');
          }
          document.getElementById('backlogJson').value = JSON.stringify(data.tasks, null, 2);
          const validation = document.getElementById('backlogValidation');
          const expectedCount = data.participantCount;
          const expectedNames = data.participants.join(', ');
          const hasVotes = data.votes && Object.keys(data.votes).length > 0;
          validation.innerHTML = `
            <div style="background: #dbeafe; border: 1px solid #93c5fd; border-radius: 8px; padding: 12px; font-size: 13px;">
              <strong>📋 Backlog restauré</strong>
              <div style="margin-top: 8px; background: #dcfce7; padding: 8px; border-radius: 4px;">
                <strong>🔐 Session:</strong> <code>${data.sessionCode}</code> (RÉUTILISÉE - même code)
              </div>
              <div style="margin-top: 8px;">
                <strong>👥 Participants EXACTS (${expectedCount}):</strong>
                <div style="margin-top: 4px; padding: 8px; background: #f3f4f6; border-radius: 4px; font-family: monospace; font-size: 12px;">
                  ${expectedNames}
                </div>
              </div>
              ${hasVotes ? `<div style="margin-top: 8px; color: #dc2626;">📊 Votes précédents détectés</div>` : ''}
              <div style="margin-top: 8px; color: #059669; padding: 8px; background: #f0fdf4; border-radius: 4px;">
                ✅ Seuls ces joueurs avec ces NOMS EXACTS peuvent se reconnecter
              </div>
            </div>
          `;
          this.restoredMetadata = {
            sessionCode: data.sessionCode,
            participantCount: data.participantCount,
            participants: data.participants
          };
          this.restoredVotes = data.votes || {};
        } else {
          document.getElementById('backlogJson').value = JSON.stringify(data, null, 2);
          document.getElementById('backlogValidation').innerHTML = '';
          this.restoredMetadata = null;
          this.restoredVotes = {};
        }
      } catch (err) {
        alert('Erreur: JSON invalide - ' + err.message);
      }
    };
    reader.readAsText(file);
  }

  /**
   * Démarre une partie à partir d'un backlog restauré
   * 
   * @function startResumedGame
   * @returns {void}
   * 
   * @description
   * - Vérifie qu'au moins un joueur est connecté
   * - Extrait métadonnées et tâches du backlog restauré
   * - Crée la session via API /api/create-session
   * - Attend que les joueurs se connectent (listener 'players-ready')
   * - Envoie le backlog après 3 secondes (timeout)
   */
  startResumedGame() {
    const connected = this.gameState.players.length;
    
    if (connected === 0) {
      alert('❌ Aucun joueur connecté. Impossible de démarrer.');
      return;
    }

    console.log('📤 this.localBacklog:', this.localBacklog);

    // ✅ EXTRAIRE LA MÉTADATA
    const metadata = this.localBacklog.metadata || {};
    const tasks = this.localBacklog.tasks || this.localBacklog;

    // ✅ DÉFINIR LE SESSION CODE
    this.sessionCode = metadata.sessionCode;
    console.log(`✅ sessionCode défini: ${this.sessionCode}`);

    // ✅ CRÉER LA SESSION D'ABORD
    fetch(`${window.location.protocol}//${window.location.hostname}:3000/api/create-session`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        sessionCode: this.sessionCode,
        creatorName: this.creatorName,
        gameMode: metadata.gameMode || this.gameMode,
        backlog: tasks,
        completedTasks: metadata.completedTasks || [],
        taskEstimates: metadata.taskEstimates || {},
        currentTaskIndex: metadata.currentTaskIndex || 0,
        currentTaskRound: metadata.currentTaskRound || 1
      })
    })
    .then(res => res.json())
    .then(data => {
      console.log('✅ Session créée:', data);
      
      // ✅ NE PAS ENVOYER LOAD-BACKLOG IMMÉDIATEMENT
      // On va attendre que les joueurs se connectent via join-session
      // et ENSUITE envoyer load-backlog
      
      // Utilise un listener pour savoir quand les joueurs sont prêts
      this.socket.once('players-ready', () => {
        console.log('✅ Tous les joueurs sont prêts, envoi du backlog');
        
        this.socket.emit('load-backlog', {
          sessionCode: this.sessionCode,
          backlog: tasks,
          isRestored: false,
          completedTasks: metadata.completedTasks || [],
          taskEstimates: metadata.taskEstimates || {},
          currentTaskVotes: metadata.currentTaskVotes || {},
          currentTaskIndex: metadata.currentTaskIndex || 0,
          currentTaskRound: metadata.currentTaskRound || 1
        });
      });

      // Envoyer le backlog après 3 secondes si on attend trop
      setTimeout(() => {
        this.socket.emit('load-backlog', {
          sessionCode: this.sessionCode,
          backlog: tasks,
          isRestored: false,
          completedTasks: metadata.completedTasks || [],
          taskEstimates: metadata.taskEstimates || {},
          currentTaskVotes: metadata.currentTaskVotes || {},
          currentTaskIndex: metadata.currentTaskIndex || 0,
          currentTaskRound: metadata.currentTaskRound || 1
        });
        console.log('✅ Timeout: load-backlog envoyé malgré tout');
      }, 3000);

    })
    .catch(error => {
      console.error('❌ Erreur lors de la création de session:', error);
      alert('❌ Erreur: Impossible de créer la session');
    });
  }


  /**
   * Attache le gestionnaire d'upload de fichier de session restaurée
   * 
   * Configure le listener sur l'input file du DOM pour
   * permettre l'upload et le parsing de fichiers de session JSON
   * 
   * @function attachFileHandler
   * @returns {void}
   * 
   * @description
   * - Récupère l'élément input file #resumeFile du DOM
   * - Attache un listener d'événement 'change'
   * - Lit et parse le fichier JSON
   * - Stocke l'intégralité du fichier dans this.localBacklog
   * - Affiche un message de statut
   */
  attachFileHandler() {
    const fileInput = document.getElementById('resumeFile');
    
    if (!fileInput) {
      console.warn('❌ Pas de resumeFile trouvé');
      return;
    }

    fileInput.addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = (event) => {
        try {
          const data = JSON.parse(event.target.result);
          console.log('📄 Fichier JSON chargé complètement:', data);
          
          // ✅ Stocker le FICHIER ENTIER (metadata + tasks)
          this.localBacklog = data;
          
          console.log('✅ this.localBacklog stocké:', this.localBacklog);
          console.log('✅ Tasks dans localBacklog:', this.localBacklog.tasks);
          
          const status = document.getElementById('resumeFileStatus');
          if (status) {
            const taskCount = data.tasks ? data.tasks.length : 0;
            status.textContent = `✅ ${taskCount} tâches chargées`;
          }
        } catch (error) {
          console.error('❌ Erreur parsing JSON:', error);
          const status = document.getElementById('resumeFileStatus');
          if (status) status.textContent = '❌ Format JSON invalide';
        }
      };
      reader.readAsText(file);
    });
  }
